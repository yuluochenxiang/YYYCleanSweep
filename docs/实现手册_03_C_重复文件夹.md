## 二、算法原理（先读懂，再动手）

### 2.1 精确判定：Merkle 签名

给每个目录算一个摘要：

```
签名(目录) = md5(
                排序后的 "本层文件名|大小" 列表
              + 排序后的 "子目录名|子目录签名" 列表
            )
```

自底向上一次算完（子孙算完了才算祖先）。于是：

- 两棵目录树**文件结构完全一致** ⇒ 签名必然相同；
- 任意深度都支持，不需要保存整棵树的清单，内存只随**目录数**线性增长；
- 子目录改名、文件大小变化，都会被算进摘要里——**不会漏判**。

### 2.2 关键坑：必须删掉「被包住的组」

`projA` 和 `projB` 完全相同时，它们里面的 `src/`、`docs/` 各自也必然相同。
如果统统报出来，**同一份冗余被记了三遍账**，可释放空间虚高三倍。

所以算法里有一个 `_drop_nested()`：**只保留没有任何成员处在别的组的子目录里的那些组**。

> 这一条是实测时踩出来的（当时 19 字节的冗余被报成 38）。别删掉它。

### 2.3 近似判定

「同一项目的 v1 / v2」不会精确相同，用三个条件放宽：

1. 总大小差 < `near_size`（默认 10%）
2. 文件数差 < `near_count`（默认 10%）
3. **顶层名字集合**重合率（Jaccard）≥ `near_overlap`（默认 80%）

顶层名字集合 = 本层文件名 + 直接子目录名。用它而不是「全部子孙文件名」，
是为了控制内存——大项目动辄几万个文件，全量求交集会很慢。

为了避免两两比对（O(n²)），先把候选**按大小排序**，再用滑动窗口：
只跟后面「大小还在容差内」的比，一旦超出就 `break`。

class DupDirThread(QThread):
    """后台：先遍历建索引，再比对。两段都能中途喊停。"""
    progress = Signal(int, str)
    finished_all = Signal(object, str)

    def __init__(self, root, min_bytes, overlap, skip, parent=None):
        super().__init__(parent)
        self.root = root
        self.min_bytes = min_bytes
        self.overlap = overlap
        self.skip = skip
        self._cancel = False

    def cancel(self):
        self._cancel = True

    def run(self):
        idx = dupdirs.scan_index(
            self.root,
            skip_names=dupdirs.DEFAULT_SKIP if self.skip else frozenset(),
            should_cancel=lambda: self._cancel,
            on_progress=lambda n, p: self.progress.emit(n, p))
        if idx is None:
            self.finished_all.emit(None, self.root)
            return
        exact, near = dupdirs.find_duplicates(
            idx, min_bytes=self.min_bytes,
            near_overlap=self.overlap / 100.0,
            should_cancel=lambda: self._cancel)
        self.finished_all.emit((exact, near, len(idx)), self.root)


class DupDirPage(QWidget):

        self.sp_ovl = QSpinBox()
        self.sp_ovl.setRange(50, 100)
        self.sp_ovl.setSingleStep(5)
        self.sp_ovl.setSuffix(" %")

        self.sp_ovl.setToolTip(
            "「大部分相同」的判定松紧：顶层文件名的重合比例达到多少才算相似")

    # ---------------- 扫描 ----------------
    def start_scan(self):
        if self.thread and self.thread.isRunning():
            return
        root = os.path.expandvars((self.root_path or "").strip())
        if not root:
            QMessageBox.information(

            QMessageBox.warning(self, "路径无效", f"目录不存在：{root}")

        self.tw_near.clear()

            root,

            int(self.sp_ovl.value()),
            self.chk_skip.isChecked(),

        self.thread.progress.connect(
            lambda n, p: self.banner.set_sub(f"已扫描 {n} 个目录 · {p}"))
        self.thread.finished_all.connect(self._on_done)
        self.thread.start()

        if self.thread:
            self.thread.cancel()
        self.btn_stop.setEnabled(False)

    def _on_done(self, res, root):
        self.btn_scan.setEnabled(True)
        self.btn_stop.setEnabled(False)
        if res is None:
            self.banner.finish("扫描失败", root)
            QTimer.singleShot(1600, self.banner.hide)
            return
        exact, near, ndirs = res
        self._fill(self.tw_exact, exact)
        self._fill(self.tw_near, near)
        ex = dupdirs.groups_total(exact)
        nr = dupdirs.groups_total(near)
        self.lb_exact.setText(
            f"完全相同（{len(exact)} 组 · 可释放 {fmt(ex)}）")
        self.lb_near.setText(
            f"大部分相同（{len(near)} 组 · 可释放 {fmt(nr)}）")
        self.lb_stat.setText(
            f"扫描 <b>{ndirs}</b> 个目录 · 完全相同 <b>{len(exact)}</b> 组"
            f"（可释放 {fmt(ex)}） · 大部分相同 <b>{len(near)}</b> 组"
            f"（可释放 {fmt(nr)}）")
        self.banner.finish("查找完成", f"{ndirs} 个目录")
        QTimer.singleShot(1600, self.banner.hide)

    def _fill(self, tw, groups):
        tw.clear()
        for g in groups:
            n = len(g["paths"])
            head = QTreeWidgetItem([
                f"{n} 份拷贝" if n > 1 else "1 份",
                fmt(g["size"] * (n - 1)),
                str(g["count"]),
            ])
            head.setData(0, Qt.UserRole, "")
            head.setData(1, Qt.UserRole, g["size"] * (n - 1))
            head.setToolTip(0, "\n".join(g["paths"]))
            head.setToolTip(1, f"留一份、删其余可释放 {fmt(g['size'] * (n - 1))}"
                               f"（{g['size'] * (n - 1):,} 字节）")
            for c in (1, 2):
                head.setTextAlignment(c, Qt.AlignRight | Qt.AlignVCenter)
            tw.addTopLevelItem(head)
            for p in g["paths"]:
                it = QTreeWidgetItem([p, fmt(g["size"]), str(g["count"])])
                it.setData(0, Qt.UserRole, p)
                it.setToolTip(0, p)
                for c in (1, 2):
                    it.setTextAlignment(c, Qt.AlignRight | Qt.AlignVCenter)
                head.addChild(it)
        tw.expandAll()

    # ---------------- 右键 ----------------
    def _menu(self, tw, pos):
        it = tw.itemAt(pos)
        if not it:
            return
        path = it.data(0, Qt.UserRole)
        m = QMenu(tw)
        if path:
            m.addAction("打开位置", lambda: self._reveal(path))
            a = m.addAction("复制路径", lambda: self._copy(path))
            a.setEnabled(True)
        else:
            m.addAction("展开当前组", lambda: it.setExpanded(True))
        m.exec(tw.viewport().mapToGlobal(pos))

    def _reveal(self, path):
        try:
            subprocess.Popen(["explorer", "/select,", os.path.normpath(path)])
        except Exception:
            pass

    def _copy(self, path):
        cb = QApplication.clipboard()
        if cb:
            cb.setText(path)

    # ---------------- 配置回调 ----------------
    def _on_min_changed(self, v):
        if self.cfg:
            self.cfg.set("dupdir_min_mb", int(v))

    def _on_ovl_changed(self, v):
        if self.cfg:
            self.cfg.set("dupdir_overlap", int(v))

    def _on_skip_changed(self, _s):
        if self.cfg:
            self.cfg.set("dupdir_skip", self.chk_skip.isChecked())

## 七、附录：`core/dupdirs.py` 完整源码

（工程里已带此文件。只有确认缺失时才用下面这份重建。）

```python
"""重复文件夹检测（零依赖）。

思路：给每棵目录树算一个 **Merkle 签名**——父目录的签名由「本层文件名+大小」
和「子目录名+子目录签名」拼出来的摘要组成，自底向上一次算完。

好处：
  - 两份目录树里的文件结构完全相同 ⇒ 签名必然相同（精确判定，不必逐个文件比）；
  - 深度无限制、不用保存整棵树的清单，内存只随目录数线性增长；
  - 子目录改名会被发现（名字参与了摘要），文件内容改了也会被发现有位移/大小变化。

近似判定（同一项目的两个版本）则用「总大小接近 + 文件数接近 + 顶层名字集合重合率高」
三条件组合，比精确判定宽松，用于捞并处理。

本模块**不依赖 Qt**，命令行模式也能用。
"""
import os
import hashlib

# 默认跳过的目录名（小写比较）。这些是典型的「文件数量炸弹」，
# 既是用户最想找的东西，也最容易淹没其它结果——所以做成一个可关掉的开关。
DEFAULT_SKIP = frozenset((
    "node_modules", "__pycache__", ".git", ".svn", ".hg", ".gradle",
    "venv", ".venv", ".idea", ".vs", ".tox", "site-packages",
    "$recycle.bin", "system volume information",
))


def scan_index(root, skip_names=DEFAULT_SKIP, max_files=400000,
               should_cancel=None, on_progress=None):
    """遍历一遍目录树，返回 {目录路径: 记录}。

    每条记录是一个 dict：
        name    目录名（小写，用于稳定排序）
        parent  父目录路径（顶层为空串）
        own     [(文件名小写, 字节数)]  —— 只含本层
        kids    [(子目录名小写, 子目录路径)]
        size    含全部子孙的总字节数
        count   含全部子孙的总文件数
        digest  Merkle 签名
        depth   层深

    返回 None 表示扫不动（根不存在 / 无权限）。
    """
    root = os.path.normpath(root)
    if not os.path.isdir(root):
        return None

    idx = {}
    nfiles = 0
    stopped = False

    for dirpath, dirnames, filenames in os.walk(root, topdown=True):
        if should_cancel and should_cancel():
            stopped = True
            break
        if skip_names:
            dirnames[:] = [d for d in dirnames if d.lower() not in skip_names]
        rec = idx.setdefault(dirpath, {"name": os.path.basename(dirpath).lower(),
                                       "parent": "", "own": [], "kids": [],
                                       "size": 0, "count": 0, "digest": "",
                                       "depth": 0})
        for d in dirnames:
            sub = os.path.join(dirpath, d)
            sub_rec = idx.setdefault(
                sub, {"name": d.lower(), "parent": "", "own": [], "kids": [],
                      "size": 0, "count": 0, "digest": "", "depth": 0})
            sub_rec["parent"] = dirpath
            rec["kids"].append((d.lower(), sub))
        for fn in filenames:
            p = os.path.join(dirpath, fn)
            try:
                sz = os.stat(p).st_size
            except OSError:
                continue
            rec["own"].append((fn.lower(), sz))
            nfiles += 1
        if on_progress and nfiles % 2000 < 1:
            on_progress(len(idx), dirpath)
        if nfiles >= max_files:
            stopped = True
            break

    _aggregate(idx, root)
    # idx 里只允许放 dict：以前往里塞过布尔值，aggregate 遍历时会当场崩
    rr = idx.get(root)
    if rr is not None:
        rr["stopped"] = stopped
    return idx


def _aggregate(idx, root):
    """自底向上：算总大小、总文件数、Merkle 签名。"""
    items = [(p, r) for p, r in idx.items()]
    for p, r in items:
        r["size"] = sum(s for _n, s in r["own"])
        r["count"] = len(r["own"])
    # 按路径长度倒序 = 深的先算（子孙的路径一定比祖先长）
    ordered = sorted(items, key=lambda t: len(t[0]), reverse=True)
    for p, r in ordered:
        r["digest"] = _digest(r["own"], r["kids"], idx)
        par = r["parent"]
        if par and par in idx:
            idx[par]["size"] += r["size"]
            idx[par]["count"] += r["count"]


def _digest(own, kids, idx):
    h = hashlib.md5()
    for name, size in sorted(own):
        h.update(("%s|%d\n" % (name, size)).encode("utf-8", "replace"))
    for kid_name, kid_path in sorted(kids):
        kr = idx.get(kid_path)
        h.update(("D|%s|%s\n" % (kid_name, kr["digest"] if kr else ""))
                 .encode("utf-8", "replace"))
    return h.hexdigest()


def _has_ancestor(path, paths):
    """path 的某个祖先目录是不是也在 paths 里。"""
    anc = os.path.dirname(path.rstrip("\\/"))
    while anc and os.path.dirname(anc) != anc:
        if anc in paths:
            return True
        anc = os.path.dirname(anc)
    return False


def _drop_nested(groups):
    """丢掉被更大的组完全包住的组。

    projA 与 projB 整体重复时，它们里面的 src/、docs/ 各自也必然重复。
    如果都报出来，同一份冗余就会被记三遍账、可释放空间虚高三倍。
    只保留「没有任何成员处在别的组的子目录里」的那些组。
    """
    allp = set()
    for g in groups:
        allp.update(g["paths"])
    out = []
    for g in groups:
        if any(not _has_ancestor(p, allp) for p in g["paths"]):
            out.append(g)
    return out


def _top_names(r):
    """顶层名字集合：本层文件名 + 直接子目录名（带前缀区分）。"""
    s = set(n for n, _sz in r["own"])
    s.update("d:" + k for k, _p in r["kids"])
    return s


def _jaccard(a, b):
    if not a or not b:
        return 0.0
    return len(a & b) / float(len(a | b))


def find_duplicates(idx, min_bytes=10 * 1024 * 1024, min_count=2,
                    near_size=0.10, near_count=0.10, near_overlap=0.80,
                    should_cancel=None):
    """在索引里找重复目录。返回 (exact_groups, near_groups)。

    每组的形状：{"paths": [...], "size": 单个的大小, "count": 单个的文件数}
    可释放空间 = size * (len(paths) - 1)。

    min_bytes / min_count：太小太碎的目录不参与比对（噪音且没价值）。
    near_size / near_count：近似的容差（比例）。
    near_overlap：顶层名字集合的最小重合率（Jaccard）。
    """
    if not idx:
        return [], []

    used = set()
    exact = []

    def eligible(r):
        return (r["size"] >= min_bytes and r["count"] >= min_count
                and r["count"] > 0 and r["digest"])

    # ---- 精确：签名相同即为完全一致
    by_digest = {}
    for path, r in idx.items():
        if not eligible(r):
            continue
        by_digest.setdefault(r["digest"], []).append(path)
    for _d, paths in by_digest.items():
        if len(paths) < 2:
            continue
        paths.sort()
        used.update(paths)
        exact.append({"paths": paths, "size": idx[paths[0]]["size"],
                      "count": idx[paths[0]]["count"]})
    exact = _drop_nested(exact)
    exact.sort(key=lambda g: g["size"] * (len(g["paths"]) - 1), reverse=True)

    # ---- 近似：大小接近的放一起滑窗比，避免全表两两比对
    cand = [(p, r) for p, r in idx.items()
            if eligible(r) and p not in used]
    cand.sort(key=lambda t: t[1]["size"])

    parent = {p: p for p, _r in cand}

    def find(x):
        while parent[x] != x:
            parent[x] = parent[parent[x]]
            x = parent[x]
        return x

    def union(a, b):
        ra, rb = find(a), find(b)
        if ra != rb:
            parent[rb] = ra

    sets = {}
    for p, r in cand:
        sets[p] = _top_names(r)

    for i, (pa, ra) in enumerate(cand):
        if should_cancel and should_cancel():
            break
        ceili = ra["size"] * (1.0 + near_size)
        for pb, rb in cand[i + 1:]:
            if rb["size"] > ceili:
                break                       # 后面的只会更大，停
            if abs(ra["size"] - rb["size"]) > ra["size"] * near_size:
                continue
            if abs(ra["count"] - rb["count"]) > max(1, ra["count"] * near_count):
                continue
            if _jaccard(sets[pa], sets[pb]) < near_overlap:
                continue
            union(pa, pb)

    groups = {}
    for p, _r in cand:
        root_p = find(p)
        groups.setdefault(root_p, []).append(p)

    near = []
    for _root, paths in groups.items():
        if len(paths) < 2:
            continue
        paths.sort()
        near.append({"paths": paths, "size": idx[paths[0]]["size"],
                     "count": idx[paths[0]]["count"]})
    near = _drop_nested(near)
    near.sort(key=lambda g: g["size"] * (len(g["paths"]) - 1), reverse=True)
    return exact, near


def groups_total(groups):
    """一组列表总共能释放多少字节（每组留一份，其余算可释放）。"""
    return sum(g["size"] * (len(g["paths"]) - 1) for g in groups)
```
