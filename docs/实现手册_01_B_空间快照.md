## 二、为什么这么做（理解后不容易写歪）

- **`core/cache.py` 已经有一整套 SQLite 设施**（锁、短连接、异常静默）。快照**复用同一个 db 文件**，另开一张表，不新建数据库文件。这样设置页那套「打开位置」逻辑直接沿用。
- **不要给快照加 TTL**。`dir_cache` 有 7 天 TTL 是因为它缓存的是会失效的汇总；而快照的价值恰恰在于**它足够旧**。加了 TTL 这个功能就废了。
- **中止的扫描不记快照**。用户中途按了停止，树是残缺的，大小偏小；拿它当基准，下次完整扫描会虚报一大堆「暴涨」。

**Step 2**：在 `_ensure_schema(conn)` 函数**之后**（约 69 行处）新增四个函数。
（不要改动 `_ensure_schema` 本身，也不要动 `get_dir` / `put_dir` / `clear` / `stats`。）

```python
def _ensure_snapshot_schema(conn):
    conn.execute(
        """CREATE TABLE IF NOT EXISTS snapshot (
            root   TEXT NOT NULL,
            ts     REAL NOT NULL,
            path   TEXT NOT NULL,
            size   INTEGER NOT NULL,
            fcount INTEGER NOT NULL,
            PRIMARY KEY (root, ts, path)
        )"""
    )
    conn.execute(
        "CREATE INDEX IF NOT EXISTS idx_snap_root_ts ON snapshot(root, ts)")
    conn.commit()


def set_snapshot_keep(n):
    """设置每个扫描根保留多少批快照（立即生效，下次写入时淘汰）。"""
    global _snap_keep
    try:
        n = int(n)
    except (TypeError, ValueError):
        return
    _snap_keep = max(1, min(500, n))


def snapshot_batches(root, limit=None):
    """这个扫描根下已有的批次时间戳（新的在前）。库不可用时返回 []。"""
    key = _norm_key(root)
    with _lock:
        conn = _connect()
        if conn is None:
            return []
        try:
            _ensure_snapshot_schema(conn)
            rows = conn.execute(
                "SELECT DISTINCT ts FROM snapshot WHERE root=? ORDER BY ts DESC",
                (key,)).fetchall()
        except sqlite3.Error:
            rows = []
        finally:
            conn.close()
    ts = [r[0] for r in rows]
    return ts[:limit] if limit else ts
```

**Step 3**：继续追加写入与对比：

```python
def put_snapshot(root, rows):
    """写一批快照，返回本次批次的时间戳。

    rows : [(path, size, count)]，一条一个目录。
    写完按保留份数滚动淘汰旧批次——这是本表唯一可能失控的地方。
    """
    key = _norm_key(root)
    ts = time.time()
    data = [(key, ts, _norm_key(p), int(s), int(c)) for p, s, c in rows]
    with _lock:
        conn = _connect()
        if conn is None:
            return ts
        try:
            _ensure_snapshot_schema(conn)
            conn.executemany(
                "INSERT OR REPLACE INTO snapshot (root, ts, path, size, fcount) "
                "VALUES (?,?,?,?,?)", data)
            stale = conn.execute(
                "SELECT DISTINCT ts FROM snapshot WHERE root=? "
                "ORDER BY ts DESC LIMIT -1 OFFSET ?", (key, _snap_keep)).fetchall()
            for (old,) in stale:
                conn.execute(
                    "DELETE FROM snapshot WHERE root=? AND ts=?", (key, old))
            conn.commit()
        except sqlite3.Error:
            pass
        finally:
            conn.close()
    return ts


def diff_snapshots(root, ts_old, ts_new, limit=200):
    """对比两个批次，返回 [(path, old_size, new_size, delta)]，按涨幅降序。

    只报「两次都在」的路径：新增和删除的目录不算增长。
    大小没变的直接跳过，免得用户看一屏 0。
    """
    key = _norm_key(root)
    with _lock:
        conn = _connect()
        if conn is None:
            return []
        try:
            _ensure_snapshot_schema(conn)
            old_d = dict(conn.execute(
                "SELECT path, size FROM snapshot WHERE root=? AND ts=?",
                (key, ts_old)).fetchall())
            new_d = dict(conn.execute(
                "SELECT path, size FROM snapshot WHERE root=? AND ts=?",
                (key, ts_new)).fetchall())
        except sqlite3.Error:
            return []
        finally:
            conn.close()
    out = []
    for p, ns in new_d.items():
        old_sz = old_d.get(p)
        if old_sz is None or old_sz == ns:
            continue
        out.append((p, old_sz, ns, ns - old_sz))
    out.sort(key=lambda x: x[3], reverse=True)
    return out[:limit]
```

> ⚠️ **变量名不要用 `os`** —— 会把已经 import 的标准库 `os` 盖掉，后面 `os.path` 全挂。
> 上面代码里用的是 `old_sz`，照抄别改。

```python
def snapshot_stats():
    """返回 (批次数, 行数)。批次数按 (root, ts) 去重计。"""
    with _lock:
        conn = _connect()
        if conn is None:
            return 0, 0
        try:
            _ensure_snapshot_schema(conn)
            b = conn.execute(
                "SELECT COUNT(*) FROM (SELECT DISTINCT root, ts FROM snapshot)"
            ).fetchone()[0]
            n = conn.execute("SELECT COUNT(*) FROM snapshot").fetchone()[0]
        except sqlite3.Error:
            b, n = 0, 0
        finally:
            conn.close()
    return b, n


def clear_snapshots():
    """清空全部空间快照。只删 snapshot 表，绝不碰 dir_cache。"""
    with _lock:
        conn = _connect()
        if conn is None:
            return
        try:
            conn.execute("DROP TABLE IF EXISTS snapshot")
            conn.commit()
        except sqlite3.Error:
            pass
        finally:
            conn.close()
```

**Step 5**：确保 `clear()` 里**只有** `DROP TABLE IF EXISTS dir_cache`，不要把 snapshot 也删了。
（现在的 `clear()` 已经只删 `dir_cache`，不用改，只是提醒别顺手加了。）

```python
    # ---------------- 空间快照（B）----------------
    def _save_snapshot(self, tree):
        """把这次扫描的每个目录占用留一份，并记住「上一次」的批次时间戳。

        注意顺序：先记下旧的批次，再写入新的——反过来的话 prev 会等于自己，
        对比结果永远全是 0。
        """
        root = tree.root
        prev_list = scan_cache.snapshot_batches(root, limit=1)
        prev = prev_list[0] if prev_list else None
        rows = [(nd.path, nd.agg_size, nd.agg_count)
                for nd in tree.nodes.values() if nd.agg_size > 0]
        cur = None
        if rows:
            scan_cache.set_snapshot_keep(int(self.cfg.get("snap_keep", 30) or 30)
                                         if self.cfg else 30)
            cur = scan_cache.put_snapshot(root, rows)
        self._snap_prev = prev
        self._snap_cur = cur

    def _clear_snap_top(self):
        """清掉「增长最快」那几个按钮，防止多次扫描叠加堆积。"""
        lay = self.snap_top_lay
        while lay.count():
            it = lay.takeAt(0)
            wd = it.widget()
            if wd is not None:
                wd.setParent(None)
                wd.deleteLater()

    def _refresh_snap_card(self):
        """刷新顶部「空间变化」卡片。没有任何一处允许留空白文案。"""
        self._clear_snap_top()
        if not self.tree:
            self.box_snap.hide()
            return
        prev = self._snap_prev
        cur = self._snap_cur
        if not prev or not cur:
            self.lb_snap.setText(
                "已记录本次扫描结果，<b>下次再扫一次</b>就能看出哪些目录在变大。")
            self.box_snap.show()
            return
        rows = scan_cache.diff_snapshots(self.tree.root, prev, cur, limit=200)
        grown = [r for r in rows if r[3] > 0]
        up = sum(r[3] for r in grown)
        down = -sum(r[3] for r in rows if r[3] < 0)
        days = max(1, int(round((cur - prev) / 86400.0)))
        dt = time.strftime("%m月%d日", time.localtime(prev))
        if up <= 0:
            self.lb_snap.setText(
                f"相比 {dt}（{days} 天前），<b>没有变大的目录</b>"
                + (f"，反倒少了 {fmt(down)}。" if down > 0 else "。"))
            self.box_snap.show()
            return
        txt = (f"相比 {dt}（{days} 天前），<b>{short_path(self.tree.root)}</b> "
               f"增加 <b>{fmt(up)}</b>")
        if down > 0:
            txt += f"，同时减少 {fmt(down)}"
        self.lb_snap.setText(txt)
        for p, _osz, _nsz, delta in grown[:3]:
            btn = QPushButton(f"{os.path.basename(p.rstrip('\\/')) or p} +{fmt(delta)}")
            btn.setObjectName("mini")
            btn.setToolTip(p)
            btn.clicked.connect(lambda _c=False, p=p: self._show_detail(p))
            self.snap_top_lay.addWidget(btn)
        self.box_snap.show()
```

> ⚠️ **lambda 闭包陷阱**：必须写成 `lambda _c=False, p=p: ...`。
> 写成 `lambda: self._show_detail(p)` 的话，三个按钮点出来全是最后一个目录
> （闭包捕获的是变量本身，不是当时的值）。

## 五、常见错误（本卷专属）

| 现象 | 原因 | 修法 |
| --- | --- | --- |
| 三个增长按钮点了都跳同一个目录 | lambda 没绑定当时的值 | 写成 `lambda _c=False, p=p: ...` |
| 对比结果永远是空的（`"没有变大的目录"`） | 先 `put_snapshot` 再取 prev | 顺序必须是**先取旧批次，再写入新批次** |
| 报错 `os.path has no attribute basename` 之类 | 用了 `os` 当变量名遮住标准库 | 把局部变量改名成 `old_sz` 之类 |
| 卡片一直不出现 | 忘了在 `_on_done` 调 `_refresh_snap_card()`，或 `box_snap` 没 `show()` | 检查 Step 5 / Step 3 最后一行的 `show()` |
